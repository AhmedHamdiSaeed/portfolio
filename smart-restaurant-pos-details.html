Sahm Smart Restaurant POS & KDS

github.com/AhmedHamdiSaeed/SmartRestaurantPOS

A high-performance Angular 18 restaurant POS dashboard designed for intense dining room workflows — featuring real-time kitchen monitors, AI meal suggestions, and zero-downtime offline transaction queuing.

View on GitHub Next Project

Implements a 4-column reactive Kanban order board (New, In Kitchen, Ready, Served) with drag-and-drop state transitions, dynamic priority indicators, and live order simulation backed by NgRx SignalStore.

Features an AI Assistant panel with character-level response streaming, recommending optimal food pairings, allergen alerts, and item modifiers with lazy-loading (@defer) and graceful failure recovery.

Built with an offline-first transaction queue using localStorage and exponential backoff retry policies. Includes a dedicated Kitchen Display System (KDS) featuring SVG load gauges, sparkline rush trends, and station bottleneck monitors.

// Reactive State Management with Angular Signals & NgRx SignalStore
export const OrderStore = signalStore(
  { providedIn: 'root' },
  withState<OrderState>({
    orders: [],
    activeFilter: 'all',
    isLoading: false,
    isSyncing: false
  }),
  withComputed((store) => ({
    kitchenOrders: computed(() => 
      store.orders().filter(o => o.status === OrderStatus.InKitchen)
    ),
    readyOrdersCount: computed(() => 
      store.orders().filter(o => o.status === OrderStatus.Ready).length
    )
  })),
  withMethods((store, orderService = inject(OrderService)) => ({
    advanceOrderStatus(orderId: string, nextStatus: OrderStatus) {
      patchState(store, {
        orders: store.orders().map(o => o.id === orderId ? { ...o, status: nextStatus } : o)
      });
    }
  }))
);
// Streaming AI Upselling & Suggestions with RxJS & Delays
@Injectable({ providedIn: 'root' })
export class AiRecommendationService {
  streamUpsellSuggestions(cartItems: CartItem[]): Observable<string> {
    const prompt = this.buildContextPrompt(cartItems);
    return this.fetchAiStream(prompt).pipe(
      concatMap(chunk => from(chunk.split(''))),
      concatMap(char => of(char).pipe(delay(18))),
      scan((acc, char) => acc + char, ''),
      catchError(() => of('Pair with signature Truffle Fries & Mint Lemonade.'))
    );
  }
}
// Offline Resilience & Indexed Queue Synchronization
export class OfflineQueueService {
  private readonly STORAGE_KEY = 'sahm_pos_offline_queue';

  enqueueTransaction(order: OrderPayload): void {
    const queue = this.getQueue();
    queue.push({ ...order, enqueuedAt: Date.now() });
    localStorage.setItem(this.STORAGE_KEY, JSON.stringify(queue));
  }

  async syncPending(): Promise<void> {
    const queue = this.getQueue();
    for (const item of queue) {
      await this.postWithRetry(item);
    }
    localStorage.removeItem(this.STORAGE_KEY);
  }
}
